game ColorMatch {
    players 2 .. 4
    deck {
        card NumberCard {
            color { red, blue, green, yellow }
            value 0 .. 9
        }
        card SkipCard {
            color { red, blue, green, yellow }
            effect skip_next_player
        }
        card DrawTwoCard {
            color { red, blue, green, yellow }
            effect draw 2
        }
        card WildCard {
            effect change_color
        }
    }
    hand 7
    play_rule {
        allow if same_color
        allow if same_value
        allow if wild
    }
    rules {
        if played SkipCard {
            skip_next_player
        }
        if played DrawTwoCard {
            draw 2
        }
        if played WildCard {
            choose_color
        }
    }
    turn {
        if cannot_play {
            draw 1
        }
        else {
            may play 1
            may draw 1
        }
    }
    actions {
        draw
        play
        choose_color
        skip_next_player
    }
    win if empty_hand
}
